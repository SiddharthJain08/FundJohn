# SP-3.1 Phase C — Crypto Redeploy + Regime Gating Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A confirmed crypto regime transition triggers a dedicated crypto-only redeploy that adjusts crypto positions (margin-funded, 24/7) and never touches equity; crypto strategies gate on the crypto regime.

**Architecture:** A new `scripts/redeploy_crypto.py` drives a new `src/execution/crypto_redeploy_sizer.py` that reuses the regime-blended weight math (`strategy_weights.load_current` → ticker-weight → normalize) but scoped to crypto strategies and with broker positions **filtered to crypto tickers before any netting** — so equity positions are structurally invisible. `engine.run_strategies` selects the regime per-strategy by `instrument_class`. The Phase B detector spawns the redeploy on a confirmed transition. All gated OFF (`OPENCLAW_CRYPTO_REDEPLOY`), inert until Phase D mints a live crypto strategy.

**Tech Stack:** Python 3 (psycopg2, the Alpaca CLI), pytest + unittest.mock.

**Spec:** `docs/superpowers/specs/2026-05-26-sp3.1-phase-c-crypto-redeploy-design.md`

**Safety invariant (the whole point):** the crypto redeploy NEVER enumerates or touches equity positions. Funded by 4x margin; the next daily cycle corrects total leverage. The load-bearing test (Task 2) runs the crypto sizer against a mixed SPY+BTC broker book and asserts zero equity orders.

```python
"""SP-3.1 Phase C: crypto-scoped position sizer for the dedicated crypto redeploy.

Reuses the regime-blended weight model (strategy_weights.load_current → daily_weight
→ ticker_weight → normalize) but (a) scoped to crypto strategies, (b) sized to crypto's
PROPORTIONAL weight slice of the leverage budget (margin-funded, not the full λ×NAV),
and (c) **broker positions filtered to crypto tickers before any delta/orphan logic**,
so equity positions are structurally invisible to this path (the load-bearing safety
invariant). Mirrors regime_blended_sizer._sharpe_cadence_path steps 1-5; equity orphan
logic is never invoked here."""
from __future__ import annotations
import logging

logger = logging.getLogger(__name__)

LAMBDA_GLOBAL = 2.0  # matches the equity sizer's default leverage cap


def _is_crypto_ticker(t: str) -> bool:
    return bool(t) and t.strip().upper().endswith('-USD')


def _dir_to_int(d) -> int:
    s = str(d or '').lower()
    return 1 if s in ('long', 'buy', '1') else (-1 if s in ('short', 'sell', '-1') else 0)


def size_crypto_positions(account_state: dict, crypto_regime_state: dict, *,
                          broker_loader=None, weights_loader=None, signals_loader=None,
                          crypto_strategy_ids=None, all_live_weight_sum=None) -> list[dict]:
    """Return crypto order dicts (ticker, direction, notional_usd, strategy_id, close_only).
    All loaders are injectable for testing; defaults wire to the live sources."""
    regime = crypto_regime_state.get('state')
    if not regime:
        logger.info('[crypto_sizer] no crypto regime — no orders')
        return []
    nav = float(account_state.get('equity') or 0.0)
    if nav <= 0:
        return []

    # 1. crypto-strategy weights for the crypto regime
    if weights_loader is None:
        from execution import strategy_weights as _sw
        weights_loader = _sw.load_current
    rows = weights_loader(regime) or []
    if crypto_strategy_ids is None:
        from strategies.instrument_class import instrument_class_for
        crypto_strategy_ids = {r['strategy_id'] for r in rows
                               if instrument_class_for(r['strategy_id']) == 'crypto'}
    weight_by_strat = {r['strategy_id']: float(r['daily_weight']) for r in rows
                       if r['strategy_id'] in crypto_strategy_ids}
    if not weight_by_strat:
        return []

    # 2. crypto signals (latest per (strategy,ticker)); injectable
    if signals_loader is None:
        from execution import strategy_weights as _sw
        signals_loader = lambda rg, wbs: _sw.load_active_signals(rg, wbs) if hasattr(_sw, 'load_active_signals') else []
    signals = signals_loader(regime, weight_by_strat) or []

    # 3. ticker_weight = Σ daily_weight × direction (crypto strategies only)
    ticker_w: dict[str, float] = {}
    for s in signals:
        sid, tkr = s.get('strategy_id'), s.get('ticker')
        if sid not in weight_by_strat or not _is_crypto_ticker(tkr or ''):
            continue
        ticker_w[tkr] = ticker_w.get(tkr, 0.0) + weight_by_strat[sid] * _dir_to_int(s.get('direction'))
    ticker_w = {t: w for t, w in ticker_w.items() if w != 0.0}

    # 4. crypto budget = crypto's PROPORTIONAL weight slice of λ×NAV (margin-funded,
    #    relative to the prior cycle's full live-weight set). all_live_weight_sum is
    #    Σ daily_weight over ALL live strategies (crypto + equity) — the denominator.
    if all_live_weight_sum is None:
        all_live_weight_sum = sum(float(r['daily_weight']) for r in rows) or 1.0
    crypto_weight_sum = sum(weight_by_strat.values())
    crypto_share = (crypto_weight_sum / all_live_weight_sum) if all_live_weight_sum else 0.0
    crypto_budget = LAMBDA_GLOBAL * nav * crypto_share
    abs_w = sum(abs(w) for w in ticker_w.values())
    target_usd = {t: (w * crypto_budget / abs_w) for t, w in ticker_w.items()} if abs_w else {}

    # 5. delta-rebalance vs CRYPTO-ONLY broker positions (the safety filter)
    if broker_loader is None:
        from execution.regime_blended_sizer import _load_broker_positions_usd
        broker_loader = _load_broker_positions_usd
    broker_crypto = {t: v for t, v in (broker_loader() or {}).items() if _is_crypto_ticker(t)}

    orders: list[dict] = []
    for tkr, tgt in target_usd.items():
        cur = broker_crypto.get(tkr, 0.0)
        delta = tgt - cur
        if abs(delta) < 1.0:
            continue
        orders.append({'ticker': tkr, 'direction': 'long' if delta > 0 else 'short',
                       'notional_usd': abs(delta), 'strategy_id': 'crypto_redeploy',
                       'close_only': False})
    # orphan-close: crypto positions held but no longer in target (equity never seen)
    for tkr, cur in broker_crypto.items():
        if tkr in target_usd or cur == 0.0:
            continue
        orders.append({'ticker': tkr, 'direction': 'short' if cur > 0 else 'long',
                       'notional_usd': abs(cur), 'strategy_id': '__close_orphan_crypto__',
                       'close_only': True, 'current_usd': cur})
    return orders
```

```python
"""SP-3.1 Phase C: dedicated crypto-only redeploy. Gated on OPENCLAW_CRYPTO_REDEPLOY.
Sizes crypto positions (crypto_redeploy_sizer), submits via the executor's crypto path,
NEVER touches equity. No RTH gate (crypto 24/7). Own cooldown namespace redeploy:crypto:*."""
from __future__ import annotations
import argparse, os, sys
from datetime import date as _date
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT)); sys.path.insert(0, str(ROOT / 'src'))

COOLDOWN_TTL_S = 60 * 60
SENTINEL_TTL_S = 24 * 60 * 60


def _cooldown_key(run_date: str) -> str:
    return f'redeploy:crypto:cooldown:{run_date}'


def _sentinel_key(run_date: str, reason: str) -> str:
    return f'redeploy:crypto:fired:{run_date}:{reason}'


def main(argv=None) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument('--reason', required=True)
    ap.add_argument('--date', default=str(_date.today()))
    ap.add_argument('--dry-run', action='store_true')
    args = ap.parse_args(argv)

    if os.environ.get('OPENCLAW_CRYPTO_REDEPLOY') != '1':
        print('[crypto-redeploy] gate OFF (OPENCLAW_CRYPTO_REDEPLOY != 1) — no-op')
        return 0

    # Cooldown + sentinel (crypto namespace)
    try:
        from database.redis import get_client as _redis
        r = _redis()
        if r and r.get(_cooldown_key(args.date)):
            print('[crypto-redeploy] blocked: cooldown active'); return 0
        if r and r.get(_sentinel_key(args.date, args.reason[:80])):
            print('[crypto-redeploy] blocked: already fired this reason'); return 0
    except Exception as e:
        r = None
        print(f'[crypto-redeploy] redis unavailable ({e}) — proceeding without cooldown')

    # Account state (NAV) + crypto sizing
    from execution.crypto_redeploy_sizer import size_crypto_positions
    from regime.crypto_regime import load_crypto_regime_state
    account_state = _load_account_state()          # see helper below
    crypto_regime = load_crypto_regime_state()
    signals_loader = _make_crypto_signals_loader()  # reads execution_signals for crypto strategies
    orders = size_crypto_positions(account_state, crypto_regime, signals_loader=signals_loader)
    if not orders:
        print('[crypto-redeploy] no crypto orders (empty signals or flat) — no-op'); return 0

    if args.dry_run:
        print(f'[crypto-redeploy] DRY-RUN — would submit {len(orders)} crypto orders: '
              f'{[o["ticker"] for o in orders]}'); return 0

    # Submit via the executor's crypto path (Phase A _route_crypto_order, through execute_single)
    from execution import alpaca_executor as ae
    submitted = 0
    for o in orders:
        res = ae.execute_single(sess=None, equity=account_state.get('equity', 0.0),
                                order=o, run_date=args.date)
        if (res or {}).get('status') == 'submitted':
            submitted += 1
        print(f'[crypto-redeploy] {o["ticker"]} -> {(res or {}).get("status")}')

    if r:
        r.set(_cooldown_key(args.date), '1', ex=COOLDOWN_TTL_S)
        r.set(_sentinel_key(args.date, args.reason[:80]), '1', ex=SENTINEL_TTL_S)
    print(f'[crypto-redeploy] submitted {submitted}/{len(orders)} crypto orders')
    return 0


def _load_account_state() -> dict:
    """NAV/equity from the Alpaca account (CLI)."""
    import subprocess, json
    try:
        proc = subprocess.run(['/root/go/bin/alpaca', 'account', 'get'],
                              capture_output=True, text=True, timeout=15, check=False)
        if proc.returncode == 0:
            acct = json.loads(proc.stdout)
            return {'equity': float(acct.get('equity') or 0.0)}
    except Exception:
        pass
    return {'equity': 0.0}


def _make_crypto_signals_loader():
    """Return a signals_loader(regime, weight_by_strat) reading today's open crypto
    signals from execution_signals for the given crypto strategies."""
    def _load(regime, weight_by_strat):
        import psycopg2, psycopg2.extras
        uri = os.environ.get('POSTGRES_URI', '')
        if not uri or not weight_by_strat:
            return []
        sids = tuple(weight_by_strat.keys())
        conn = psycopg2.connect(uri)
        try:
            with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
                cur.execute(
                    "SELECT DISTINCT ON (strategy_id, ticker) strategy_id, ticker, direction "
                    "FROM execution_signals WHERE strategy_id IN %s AND status = 'open' "
                    "ORDER BY strategy_id, ticker, created_at DESC", (sids,))
                return [dict(r) for r in cur.fetchall()]
        finally:
            conn.close()
    return _load


if __name__ == '__main__':
    sys.exit(main())

def _spawn_crypto_redeploy(prior_state: str, new_state: str, date_str: str) -> str:
    """Spawn scripts/redeploy_crypto.py DETACHED (mirrors the equity intraday spawn).
    Gated: only spawns when OPENCLAW_CRYPTO_REDEPLOY=1."""
    if os.environ.get('OPENCLAW_CRYPTO_REDEPLOY') != '1':
        return 'gate_off'
    reason = f'CRYPTO_HMM_{prior_state}_{new_state}'
    log_dir = ROOT / 'logs'
    try:
        log_dir.mkdir(parents=True, exist_ok=True)
        log_fd = os.open(str(log_dir / f'redeploy_crypto_{date_str}.log'),
                         os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o644)
    except Exception:
        log_fd = _subprocess.DEVNULL
    try:
        _subprocess.Popen(
            [sys.executable, str(ROOT / 'scripts' / 'redeploy_crypto.py'),
             '--reason', reason, '--date', date_str],
            cwd=str(ROOT), stdin=_subprocess.DEVNULL, stdout=log_fd, stderr=log_fd,
            start_new_session=True, close_fds=True)
    except Exception as e:
        print(f'[crypto-regime] redeploy spawn failed: {e}')
        return 'spawn_error'
    finally:
        if isinstance(log_fd, int) and log_fd != _subprocess.DEVNULL:
            try: os.close(log_fd)
            except Exception: pass
    return 'spawned'
```
