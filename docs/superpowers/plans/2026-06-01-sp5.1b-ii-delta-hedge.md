# SP-5.1b-ii — Delta-Hedge Subsystem Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Delta-hedge live long straddle/strangle positions by emitting underlying-equity hedge targets into the SP-6 EOD compute→T+1-fill pipeline, tracked in a dedicated `option_hedge_ledger` that keeps the equity book structurally invisible to hedging.

**Architecture:** The hedge is a **target producer**, not a submitter. At the 4 PM[T] EOD compute, for each live delta-hedged option structure, sum the live chain leg deltas → write the target hedge share position (`−Σδ×100×contracts`) as a COMPUTED `execution_signals` row on the underlying (tagged `is_hedge` in `signal_params`) + upsert the ledger. The SP-6 9:30 reconcile / 3:55 sizer then fill it T+1. A hedge-ledger-exclusion hook in the production sizer subtracts the ledger's hedge qty from the broker book before delta-classification, so equity strategies on the same ticker are untouched. `current_hedge_qty` updates from real fills (`alpaca_submissions.filled_qty`).

**Tech Stack:** Python 3, psycopg2 (Postgres), Alpaca CLI, pytest, JS (daily-cycle.js / cron-schedule.js wiring). Builds on SP-5.1b-i (mleg exec, on `main`) + SP-6 Phase A (EOD seams).

**Spec:** `docs/superpowers/specs/2026-06-01-sp5.1b-mleg-straddle-strangle-design.md` (§3 EOD target-producer, §4.2 components, §5 safety).

```sql
    updated_at         TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (option_strategy_id, underlying)
);
```

```python
"""src/execution/option_hedge.py — SP-5.1b-ii delta-hedge ledger + target producer."""
from __future__ import annotations
import json

def upsert_hedge_target(cur, strategy_id, underlying, legs, contracts,
                        target_hedge_qty, as_of):
    """Upsert the ledger row's target_hedge_qty (the EOD-computed hedge), keyed by
    (option_strategy_id, underlying). current_hedge_qty is NOT touched here (it
    updates from real fills — see update_hedge_filled)."""
    cur.execute(
        """INSERT INTO option_hedge_ledger
             (option_strategy_id, underlying, structure_legs, contracts,
              target_hedge_qty, last_rehedge_date, status, updated_at)
           VALUES (%s,%s,%s::jsonb,%s,%s,%s,'active',NOW())
           ON CONFLICT (option_strategy_id, underlying) DO UPDATE SET
             structure_legs=EXCLUDED.structure_legs, contracts=EXCLUDED.contracts,
             target_hedge_qty=EXCLUDED.target_hedge_qty,
             last_rehedge_date=EXCLUDED.last_rehedge_date,
             status='active', updated_at=NOW()""",
        (strategy_id, underlying, json.dumps(legs), contracts,
         target_hedge_qty, as_of))

def load_active_hedges(cur):
    """All active ledger rows as dicts."""
    cur.execute("""SELECT option_strategy_id, underlying, structure_legs, contracts,
                          current_hedge_qty, target_hedge_qty
                   FROM option_hedge_ledger WHERE status='active'""")
    cols = [d[0] for d in cur.description]
    return [dict(zip(cols, r)) for r in cur.fetchall()]

def hedge_qty_by_underlying(cur):
    """{underlying: Σ current_hedge_qty} across active hedges — the amount the
    production sizer must EXCLUDE from the broker book (Task 6)."""
    cur.execute("""SELECT underlying, COALESCE(SUM(current_hedge_qty),0)
                   FROM option_hedge_ledger WHERE status='active' GROUP BY underlying""")
    return {u: float(q) for u, q in cur.fetchall()}

def close_hedge(cur, strategy_id, underlying):
    """Mark closed + zero the target (the next EOD emits a flatten-to-0 hedge target)."""
    cur.execute("""UPDATE option_hedge_ledger SET status='closed', target_hedge_qty=0,
                   updated_at=NOW() WHERE option_strategy_id=%s AND underlying=%s""",
                (strategy_id, underlying))
```

```python
def _leg_delta(occ, right, strike, expiry):
    """Live delta for one option leg, from the chain at its exact strike.
    Reuses SP-5.1b-i's _option_chain_greeks (snapshots[occ].greeks.delta)."""
    import datetime as _dt
    from execution.alpaca_executor import _option_chain_greeks, _spot_price
    und = ''.join(c for c in occ if c.isalpha())
    spot = _spot_price(und)
    if spot is None:
        return None
    exp = _dt.date.fromisoformat(expiry) if isinstance(expiry, str) else expiry
    band = max(0.01, abs(float(strike) - spot) / spot + 0.001)
    for (k, d, o) in _option_chain_greeks(und, exp, right, spot, band_pct=band):
        if o == occ:
            return float(d)
    return None

def compute_structure_delta(legs, contracts):
    """Net share-delta of the long structure = Σ(leg delta) × 100 × contracts.
    Returns None (fail-closed) if any leg delta can't be resolved."""
    total = 0.0
    for leg in legs:
        d = _leg_delta(leg['occ'], leg['right'], leg['strike'], leg.get('expiry'))
        if d is None:
            return None
        total += d
    return total * 100.0 * float(contracts)
```

```python
def _next_trading_day(d):
    import datetime as _dt
    nd = d + _dt.timedelta(days=1)
    while nd.weekday() >= 5:   # Sat/Sun
        nd += _dt.timedelta(days=1)
    return nd

def compute_option_hedge_targets(cur, as_of):
    """EOD[T] hedge target producer. For each active hedge: target = -net_delta;
    write a COMPUTED execution_signals row on the underlying (tagged is_hedge in
    signal_params) for the SP-6 pipeline to fill T+1, and upsert the ledger target.
    A None delta (fail-closed) skips that hedge (no row → no spurious flatten)."""
    import json
    tgt_date = _next_trading_day(as_of)
    for row in load_active_hedges(cur):
        net = compute_structure_delta(row['structure_legs'], row['contracts'])
        if net is None:
            continue   # fail-closed: leave last target; do not emit a wrong hedge
        target_shares = -net
        direction = 'LONG' if target_shares > 0 else 'SHORT'
        params = {'is_hedge': True, 'hedge_strategy_id': row['option_strategy_id'],
                  'hedge_underlying': row['underlying']}
        hedge_sid = f"__hedge__{row['option_strategy_id']}"
        cur.execute(
            """INSERT INTO execution_signals
                 (strategy_id, workspace_id, signal_date, ticker, direction,
                  position_size_pct, regime_state, signal_params, status,
                  lifecycle_state, computed_at, target_date)
               VALUES (%s, COALESCE(current_setting('app.workspace', true), 'default'),
                       %s,%s,%s,%s,%s,%s::jsonb,'open','COMPUTED',NOW(),%s)
               ON CONFLICT (strategy_id, signal_date, ticker, direction) DO UPDATE SET
                 signal_params=EXCLUDED.signal_params, lifecycle_state='COMPUTED',
                 target_date=EXCLUDED.target_date, computed_at=NOW()""",
            (hedge_sid, as_of, row['underlying'], direction, abs(target_shares),
             'HEDGE', json.dumps(params), tgt_date))
        upsert_hedge_target(cur, row['option_strategy_id'], row['underlying'],
                            row['structure_legs'], row['contracts'],
```

## Task 6: Hedge-ledger-exclusion hook in the production sizer (EQUITY-UNTOUCHED)

**Files:** Modify `src/execution/regime_blended_sizer.py` (~line 706, after `_load_broker_positions_usd()`); Test `tests/test_option_hedge_exclusion.py`.

> THE load-bearing safety task. `_classify_position_deltas` reads the FULL broker book per ticker with no per-strategy filtering (grounding §4). If the hedge holds N shares of SPY and an equity strategy also holds SPY, the sizer sees the combined position and would orphan/resize the hedge's portion as if it were equity. Fix: subtract the ledger's `current_hedge_qty` (× price) from the broker book BEFORE `_classify_position_deltas`, so the equity sizer only ever sees the equity-strategy portion. The hedge's own adjustment flows as its tagged COMPUTED target (Task 4).

```python
def _exclude_hedge_from_broker(broker, hedge_qty_by_underlying):
    """Subtract option-hedge share value from the broker book so the equity sizer
    only sees the equity-strategy portion (equity-untouched invariant). hedge_qty
    is signed shares per underlying (from option_hedge.hedge_qty_by_underlying)."""
    if not hedge_qty_by_underlying:
        return broker
    out = dict(broker)
    for und, shares in hedge_qty_by_underlying.items():
        if und in out and shares:
            px = _spot_price(und)
            if px:
                out[und] = out[und] - float(shares) * float(px)
```

```python
def update_hedge_filled(cur, run_date):
    """Set current_hedge_qty from REAL filled hedge orders for run_date. Hedge
    orders use strategy_id '__hedge__<option_sid>'. Sourced from alpaca_submissions

        """SELECT strategy_id, ticker,
                  SUM(CASE WHEN lower(side)='sell' THEN -1 ELSE 1 END * COALESCE(filled_qty,0))
           FROM alpaca_submissions
           WHERE run_date=%s AND strategy_id LIKE '__hedge__%%'
                 AND lower(broker_status)='filled'

        cur.execute(
            """UPDATE option_hedge_ledger SET current_hedge_qty=current_hedge_qty+%s,
               updated_at=NOW() WHERE option_strategy_id=%s AND underlying=%s""",
```
